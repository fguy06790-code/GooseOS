function(thyfos_test name source)
    add_executable(${name} "${source}")
    target_compile_definitions(${name} PRIVATE UNICODE _UNICODE WINVER=0x0a00 _WIN32_WINNT=0x0a00)
    target_include_directories(${name} PRIVATE
        "${STORE_OVERLAY}" "${STORE}"
        "${SETUP}" "${LIBERNT_SOURCE_DIR}/base/setup/libernt" "${DESKTOP}"
        "${LIBERNT_SOURCE_DIR}/base/shell/shellframe"
        "${LIBERNT_SOURCE_DIR}/sdk/include")
    if(MINGW)
        target_compile_definitions(${name} PRIVATE PROCESSOR_ARCHITECTURE_RISCV64=15)
        target_compile_options(${name} PRIVATE -ffunction-sections -fdata-sections)
        target_link_options(${name} PRIVATE -static -Wl,--gc-sections)
        if("WIDE" IN_LIST ARGN)
            target_link_options(${name} PRIVATE -municode)
        endif()
    elseif(MSVC)
        target_compile_options(${name} PRIVATE /Gy)
        target_link_options(${name} PRIVATE /OPT:REF)
    endif()
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE)
endfunction()
